cmake_minimum_required(VERSION 3.16)
project(blockcraft CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.0 GIT_SHALLOW 1)
FetchContent_MakeAvailable(raylib)
add_executable(blockcraft src/main.cpp src/net.cpp)
target_include_directories(blockcraft PRIVATE src)
target_link_libraries(blockcraft raylib)
if(WIN32)
  target_link_libraries(blockcraft ws2_32)
endif()
